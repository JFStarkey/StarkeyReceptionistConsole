import express from "express"; import { fileURLToPath } from "url"; import { dirname, join } from "path"; import fs from "fs"; const app = express(); const port = process.env.PORT || 5000; const __filename = fileURLToPath(import.meta.url); const __dirname = dirname(__filename); console.log("ROOT:", __dirname); const bundlePath = join(__dirname, "build", "bundle.js"); console.log("BUNDLE EXISTS:", fs.existsSync(bundlePath)); console.log("BUNDLE PATH:", bundlePath); app.use(express.static(__dirname)); app.listen(port, () => { console.log("listening on 5000"); });
